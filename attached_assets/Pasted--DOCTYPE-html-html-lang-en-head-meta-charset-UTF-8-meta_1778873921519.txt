    function animateNumber(id, target, suffix = '') {
        const el = document.getElementById(id);
        if (!el) return;
        let current = 0;
        const step = target / 60;
        const timer = setInterval(() => {
            current += step;
            if (current >= target) { current = target; clearInterval(timer); }
            el.textContent = Math.floor(current).toLocaleString() + suffix;
        }, 20);
    }

    // ==================== QUIZ ====================
    let currentQuestion = 0;
    let quizScore = 0;
    let answered = false;
    let shuffledQuiz = [];

    function initQuiz() {
        currentQuestion = 0;
        quizScore = 0;
        answered = false;
        shuffledQuiz = [...quizData].sort(() => Math.random() - 0.5).slice(0, 6);
        renderQuestion();
    }

    function renderQuestion() {
        const area = document.getElementById('quizArea');
        if (currentQuestion >= shuffledQuiz.length) {
            showResult();
            return;
        }

        const q = shuffledQuiz[currentQuestion];
        const progress = ((currentQuestion) / shuffledQuiz.length) * 100;

        area.innerHTML = `
            <div class="quiz-header">
                <div style="text-align:center">
                    <div style="font-size:1.3rem;font-weight:800;color:white">${currentQuestion + 1}/${shuffledQuiz.length}</div>
                    <div style="font-size:0.8rem;color:#888">Questions</div>
                </div>
                <div class="quiz-progress">
                    <div class="progress-bar"><div class="progress-fill" style="width:${progress}%"></div></div>
                </div>
                <div style="text-align:center">
                    <div style="font-size:1.3rem;font-weight:800;color:var(--accent)">${quizScore * 15}</div>
                    <div style="font-size:0.8rem;color:#888">Points</div>
                </div>
            </div>
            <div class="question-card">
                <div class="question-num">🌿 Question ${currentQuestion + 1}</div>
                <div class="question-text">${q.q}</div>
                <div class="options-grid">
                    ${q.options.map((opt, i) => `
                        <button class="option-btn" onclick="answerQuestion(${i}, this)">
                            <span style="color:var(--primary);margin-right:8px">${['A','B','C','D'][i]}.</span> ${opt}
                        </button>
                    `).join('')}
                </div>
            </div>
            <div id="factBox"></div>
        `;
        answered = false;
    }

    function answerQuestion(selected, btn) {
        if (answered) return;
        answered = true;
        const q = shuffledQuiz[currentQuestion];
        const buttons = document.querySelectorAll('.option-btn');

        buttons.forEach((b, i) => {
            b.disabled = true;
            if (i === q.correct) b.classList.add('correct');
            else if (i === selected) b.classList.add('wrong');
        });

        const isCorrect = selected === q.correct;
        if (isCorrect) {
            quizScore++;
            showNotification('✅', 'Correct Answer!', '+15 points earned');
            addPoints(15, '🧠 Correct Quiz Answer');
        } else {
            showNotification('❌', 'Wrong Answer!', 'Better luck next time!');
        }

        document.getElementById('factBox').innerHTML = `
            <div style="background:${isCorrect ? 'rgba(46,204,113,0.1)' : 'rgba(231,76,60,0.1)'};border:1px solid ${isCorrect ? 'var(--primary)' : 'var(--danger)'};border-radius:15px;padding:20px;margin-bottom:20px">
                <p style="font-weight:700;color:${isCorrect ? 'var(--primary)' : 'var(--danger)'};margin-bottom:8px">${isCorrect ? '🎉 Correct!' : '💡 Did you know?'}</p>
                <p style="color:#aaa;line-height:1.6">${q.fact}</p>
            </div>
            <div style="text-align:center">
                <button class="btn btn-primary" onclick="nextQuestion()">
                    ${currentQuestion < shuffledQuiz.length - 1 ? '➡️ Next Question' : '🏁 See Results'}
                </button>
            </div>
        `;
    }

    function nextQuestion() {
        currentQuestion++;
        renderQuestion();
    }

    function showResult() {
        const area = document.getElementById('quizArea');
        const pct = Math.round((quizScore / shuffledQuiz.length) * 100);
        const emoji = pct >= 80 ? '🏆' : pct >= 60 ? '🌟' : pct >= 40 ? '👍' : '📚';
        const msg = pct >= 80 ? 'Eco Expert!' : pct >= 60 ? 'Great Job!' : pct >= 40 ? 'Good Effort!' : 'Keep Learning!';
        const totalPoints = quizScore * 15;

        area.innerHTML = `
            <div class="quiz-result">
                <div class="result-emoji">${emoji}</div>
                <div class="result-title">${msg}</div>
                <div class="result-score">${pct}%</div>
                <p style="color:#888;margin:15px 0">${quizScore} out of ${shuffledQuiz.length} questions correct</p>
                <div style="background:rgba(46,204,113,0.1);border-radius:15px;padding:20px;margin:25px 0;border:1px solid rgba(46,204,113,0.2)">
                    <div style="font-size:2rem;color:var(--accent);font-weight:800">+${totalPoints} Points Earned! 🎉</div>
                    ${pct >= 80 ? '<div style="color:var(--primary);margin-top:10px;font-weight:600">🏅 Quiz Master Badge Unlocked!</div>' : ''}
                </div>
                <div style="display:flex;gap:15px;justify-content:center;flex-wrap:wrap">
                    <button class="btn btn-primary" onclick="initQuiz()">🔄 Try Again</button>
                    <button class="btn btn-outline" onclick="showSection('leaderboard')">🏆 Leaderboard</button>
                </div>
            </div>
        `;

        if (pct >= 80) showModal('🏅 Achievement Unlocked!', '<div style="text-align:center"><div style="font-size:4rem;margin-bottom:20px">🧠</div><p style="color:#888;line-height:1.7">You scored '+pct+'% on the Eco Quiz! You\'ve earned the <strong style="color:var(--primary)">Quiz Master Badge</strong> and <strong style="color:var(--accent)">'+totalPoints+' points</strong>!</p></div>');
